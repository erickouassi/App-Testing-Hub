# Disallow indexing of system logic, local scripts, or mock data files
Disallow: /api/           # Block search crawlers from reading backend API payloads directly
Disallow: /js/            # Prevent direct indexing of your independent script engines
Disallow: /css/           # Protect core asset styles files from separate crawler indexing
Disallow: /*.json$        # Prevent indexing of metadata config or local pro-tip files
Disallow: /app.html?slug= # Prevent search bots from bloating crawl budgets with dynamic query strings
